cmake_minimum_required(VERSION 3.24)
project(DDSStorageBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
get_filename_component(IMAGECODEC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LIBDDS_SOURCE_DIR "${IMAGECODEC_ROOT}/External/libdds" CACHE PATH "libdds source")
set(CODECDDS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Source" CACHE PATH "CodecDDS source")
set(IMAGECODEC_INCLUDE_DIR "${IMAGECODEC_ROOT}/ImageCodec/Include" CACHE PATH "ImageCodec headers")
option(DDS_BASELINE "Build the pre-storage API baseline" OFF)
set(LIBDDS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBDDS_SOURCE_DIR}" libdds)
set(LLUTILS_SOURCE_DIR "${IMAGECODEC_ROOT}/../LLUtils" CACHE PATH "Workspace LLUtils checkout")
if(NOT EXISTS "${LLUTILS_SOURCE_DIR}/CMakeLists.txt")
    set(LLUTILS_SOURCE_DIR "${IMAGECODEC_ROOT}/External/LLUtils")
endif()
add_subdirectory("${LLUTILS_SOURCE_DIR}" llutils)
find_package(Threads REQUIRED)
add_executable(dds_storage_benchmark Benchmark.cpp)
target_include_directories(dds_storage_benchmark PRIVATE "${CODECDDS_SOURCE_DIR}" "${IMAGECODEC_INCLUDE_DIR}")
target_link_libraries(dds_storage_benchmark PRIVATE libdds::libdds LLUtils::LLUtils Threads::Threads)
if(WIN32)
    target_compile_definitions(dds_storage_benchmark PRIVATE UNICODE _UNICODE)
endif()
if(DDS_BASELINE)
    target_compile_definitions(dds_storage_benchmark PRIVATE DDS_BASELINE=1)
endif()

add_executable(dds_pool_benchmark Pool.cpp)
target_include_directories(dds_pool_benchmark PRIVATE "${CODECDDS_SOURCE_DIR}" "${IMAGECODEC_INCLUDE_DIR}")
target_link_libraries(dds_pool_benchmark PRIVATE libdds::libdds LLUtils::LLUtils Threads::Threads)
if(WIN32)
    target_compile_definitions(dds_pool_benchmark PRIVATE UNICODE _UNICODE)
endif()
